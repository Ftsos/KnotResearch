\section{Identities}

We have: 
\begin{align}
  g_4(K) &= g_4(K^r) \\
  d(K, J) &= g_4(\bar{K}, \bar{J}) \\
  d(K^r, J^r) &= d(K,J)\\
  d(K^r, J) &= d(K, J^r)
\end{align}


\subsection*{1. Equivalence Class: $K \# J$}
\[
\begin{tikzcd}[row sep=huge, column sep=huge]
K \# J \arrow[r, "\text{---}"] 
       & \overline{K} \# \overline{J} \arrow[d, "r"] \\
K^r \# J^r \arrow[u, "r"] 
       & \overline{K^r} \# \overline{J^r} \arrow[l, "\text{---}"]
\end{tikzcd}
\]

\subsection*{2. Equivalence Class: $K \# \overline{J}$}
\[
\begin{tikzcd}[row sep=huge, column sep=huge]

K^r \# \overline{J^r} \arrow[u, "r"] 
                  & \overline{K^r} \# J^r \arrow[l, "\text{---}"]
\end{tikzcd}
