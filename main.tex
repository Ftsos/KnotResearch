\section{Identities}

We have: 
\begin{align}
  g_4(K) &= g_4(K^r) \\
  d(K, J) &= g_4(\bar{K}, \bar{J}) \\
  d(K^r, J^r) &= d(K,J)\\
  d(K^r, J) &= d(K, J^r)
\end{align}

\section{A Lower Bound}

We know that the Cobordism Distance is subadditive, i.e.
$g_4(K\#J) \leq g_4(K) + g_4(J)$. From here we can show
\begin{align*}
    g_4(K) &= g_4(K \# \overline{J} \# J) \\
    \implies g_4(K) &\leq g_4(K \# \overline{J}) + g_4(J) \\
    \implies g_4(K) - g_4(J) &\leq g_4(K \# \overline{J}) \quad \blacksquare \\
\end{align*}
Similarly,
\begin{align*}
    g_4(J) &= g_4(J \# \overline{K} \# K) \\
    \implies g_4(J) &\leq g_4(J \# \overline{K}) + g_4(K) \\
    \implies g_4(J) - g_4(K) &\leq g_4(J \# \overline{K}) \\
    &= g_4(\overline{J \# \overline{K}}) \\
    &= g_4(\overline{J} \# K) \\
    &= g_4(K \# \overline{J}) \quad \blacksquare
\end{align*}
Thus,
\begin{align*}
    \left|\, g_4(K) - g_4(J) \,\right| \leq g_4(K \# \overline{J}) \leq g_4(K) + g_4(J) \quad \square
\end{align*}
